### other analyses

###		data/atac_open_coverage.txt
###		data/sample_totals.txt

#############
# 1. Initial data prep
#############

## ATAC-seq open regions from Palfy and Liu, C&R coverage (analysis2/pca_cov)
data <- read.table("data/atac_open_coverage.txt", header=TRUE)
names(data) <- tolower(names(data))

## Center, Left, right regions
left_data <- data[,grepl("_left", names(data))]
right_data <- data[,grepl("_right", names(data))]

## Swap left/right so that right summed cov is > left
## Only considering non Igg samples
non_igg_right <- right_data[,!grepl('igg', names(right_data))]
non_igg_left <- left_data[,!grepl('igg', names(left_data))]

orientation <- rowSums(non_igg_right) - rowSums(non_igg_left)

inferred_plus_strand <- subset(data, orientation >= 0)
new_names1 <- gsub("_left", "_up", names(data))
new_names1 <- gsub("_right", "_down", new_names1)
names(inferred_plus_strand) <- new_names1
inferred_plus_strand$inferred_strand <- "+"

inferred_minus_strand <- subset(data, orientation < 0)
new_names2 <- gsub("_left", "_down", names(data))
new_names2 <- gsub("_right", "_up", new_names2)
names(inferred_minus_strand) <- new_names2

## swap the columns
inferred_minus_strand <- inferred_minus_strand[,new_names1]
inferred_minus_strand$inferred_strand <- "-"

data2 <- rbind(inferred_plus_strand, inferred_minus_strand)


## Normalize center coverage to 500bp (center = atac peak, so variable length)
center_data <- data2[,grepl("_center", names(data2))]
atac_open_lengths <- data2$end - data2$start
center_data <- round(500 * center_data / atac_open_lengths, 3)

up_data <- data2[,grepl("_up", names(data2))]
down_data <- data2[,grepl("_down", names(data2))]

data3 <- cbind(data2[,1:7], data2[,'inferred_strand', drop=F], center_data, up_data, down_data)

write.table(data3, 'output/atac_open_coverage_reshaped.txt', row.names=F, quote=F, sep='\t')


#############
# 2. Data prep for pooled samples (Fig 1D)
#############

## Reconstruct the data table with pooled replicates
## Normalize to log2 RPKM, smoothing factor 1
## For Igg, pool igg and igg_dmso (rename igg_dmso_1 and igg_dmso_2 to igg_d1 and igg_d2)

names(data3) <- gsub('igg_dmso_', 'igg_d', names(data3))
names(center_data) <- gsub('igg_dmso_', 'igg_d', names(center_data))

norm_factors <- read.table('data/sample_totals.txt', header=T)

## Adjust the names to match current analysis
norm_factors$Sample_Name <- tolower(norm_factors$Sample_Name)
norm_factors$Sample_Name <- gsub('igg_dmso_', 'igg_d', norm_factors$Sample_Name)
norm_factors <- subset(norm_factors, Sample_Name %in% gsub("_center", "", names(data3)), select = c('Sample_Name', 'PCR_Dedup_1'))

## Pool norm factors across replicates
norm_factors_unpooled <- norm_factors
norm_factors$Sample_Name <- gsub("_..?$", "", norm_factors$Sample_Name, perl=T)
norm_factors <- aggregate(PCR_Dedup_1 ~ Sample_Name, norm_factors, FUN = "sum")

SMOOTH <- 1
marks <- unique(gsub("_..?_center", "", names(center_data), perl=T))
pooled <- data3[,'name', drop=F]
for (mark in marks) {
	data_subset <- data3[,grepl(mark, names(data3))]
	norm <- 2 * 1e6 / norm_factors[norm_factors$Sample_Name == mark,'PCR_Dedup_1']
		
	for (region in c('up', 'center', 'down')) {
		if (ncol(data_subset) > 3) {
			pooled[,paste0(mark, '_', region)] <- log2(norm * rowSums(data_subset[,grepl(region, names(data_subset))]) + SMOOTH)
		}
		else {
			pooled[,paste0(mark, '_', region)] <- log2(norm * data_subset[,grepl(region, names(data_subset))] + SMOOTH)
		}
	}
}

	prefix <- gsub("_[0-9d]+$", "", mark, perl=T)

# 7. Prepare for future analysis on individual histone marks
#############

## Most of the informative signal is central + downstream, so
## recalculate log2 RPKM for center+down: exponentiate, undo smoothing,
## weighted average according to atac region size, resmooth, log2

data4 <- cbind(data3[,1:8], data_pca_rot[,'svm_prediction', drop=F], pooled_with_igg[,2:ncol(pooled_with_igg)])

for (sample in c('k4me1', 'k4me2', 'k4me3', 'k9ac', 'k27ac', 'h4k16ac', 'k56ac', 'k64ac', 'k122ac', 'ntac', 'igg')) {

	center_name <- paste0('dome_', sample, '_center')
	center_rpkm <- 2^(data4[,center_name])-1

	down_name <- paste0('dome_', sample, '_down')
	down_rpkm <- 2^(data4[,down_name])-1

	region_len <- data4$end-data4$start	

	center_down_name <- paste0('dome_', sample, '_centerdown')
	center_down_rpkm <- center_rpkm * (region_len / (region_len + 1000)) + down_rpkm * (1000 / (region_len + 1000))
	data4[,center_down_name] <- log2(center_down_rpkm + 1)
}

## Calculate log2 enrichment over igg based on center_down_rpkm
for (sample in c('k4me1', 'k4me2', 'k4me3', 'k9ac', 'k27ac', 'h4k16ac', 'k56ac', 'k64ac', 'k122ac', 'ntac')) {
	center_down_name <- paste0('dome_', sample, '_centerdown')
	igg_name <- 'dome_igg_centerdown'
	
	enrich_name <- paste0('dome_', sample, '_lg_enrich')
	data4[,enrich_name] <- data4[,center_down_name] - data4[,igg_name]
}

write.table(data4, 'output/atac_open_predictions.txt', row.names=F, quote=F, sep='\t')
