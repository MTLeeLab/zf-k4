## Plotting function for expression trajectory line plot

plot_kinetics = function(mat, group_name, fg_col='#005AB5', bg_col='#005AB544') {
	n <- nrow(mat[mat$contrast %in% group_name,])
	nps_down_mean <- apply(log2(mat[mat$contrast %in% group_name, 3:ncol(mat)]), 2, mean)
	nps_down_sd <- apply(log2(mat[mat$contrast %in% group_name, 3:ncol(mat)]), 2, sd)
	nps_down_margin <- qt(0.975, df=n-1) * nps_down_sd / sqrt(n)

	points(nps_down_mean, type='l', col=fg_col, lwd=1)
	polygon(c(1:length(nps_down_mean), length(nps_down_mean):1), c(nps_down_mean-nps_down_margin, rev(nps_down_mean+nps_down_margin)), col=bg_col, border=F)
	
	summary_table <- cbind(nps_down_mean, nps_down_mean-nps_down_margin, nps_down_mean+nps_down_margin)
	summary_table
}



## Data set 1: Vejnar timecourse ribo-zero, yeast spikein normalized, unique counts only
## Stratify by maternal-zygotic vs zygotic
## Maternal = >0.5 RPKM at pooled 2-cell stage ribo-zero

vej <- read.csv('../../public_data/dev_timecourse_zebrafish_vejnar_giraldez_rpkm_1_gene_all_all_v12.csv.gz', header=T)

vej <- merge(vej, miao, by.x='gene', by.y='Geneid')
vej$Vejnar_maternal <- ifelse(vej$WT.2c.R0 > 0.5, 'MZ', 'Z')


pdf('output/vejnar_kinetics.pdf', 10,15)
par(mfcol=c(3,2), pty='s')

times <- c('WT.16c', 'WT.64c', 'WT.256c', 'WT.1Kc', 'WT.oblong', 'WT.sphere', 'WT.dome')
labels <- c('16c', '64c', '256c', '1K', 'oblong', 'sphere', 'dome')


out_table <- vej[,c('gene', 'Symbol', 'Miao_MZnps_category', 'Heyn_4su_minorwave', 'Vejnar_maternal')]
names(out_table)[1] <- 'Geneid'


out_summary_table <- data.frame(Stages=labels)

for (rep in c('.B1', '.B2', '')) {
	deltaRPKM <- data.frame(Geneid = vej$gene, contrast = paste0(vej$Vejnar_maternal, vej$Miao_MZnps_category))
	
	samples <- paste0(times, '.R0', rep)
	time0 <- samples[1]
	for (time in samples) {
		deltaRPKM[,time] <- (vej[,time]+0.1) / (vej[,time0]+0.1)
	}

	for (mzgroup in c('.', 'MZ', 'Z')) {
		mat <- subset(deltaRPKM, grepl(paste0('^', mzgroup), deltaRPKM$contrast, perl=T))
		down_keywords <- unique(mat$contrast)[grep('Down', unique(mat$contrast))]
		unaff_keywords <- unique(mat$contrast)[grep('Un', unique(mat$contrast))]


		plot(0,0,xlim=c(1,length(labels)), ylim=c(-1,5), xlab='Stage', ylab='log2 Increase', axes=F, main = paste('R0', mzgroup, rep))
		axis(1, at=1:length(labels), labels=labels)
		axis(2)
		down_table <- plot_kinetics(mat, down_keywords)
		unaff_table <- plot_kinetics(mat, unaff_keywords, '#DC3220', bg_col='#DC322044')	

		down_table <- data.frame(down_table)
		unaff_table <- data.frame(unaff_table)
		
		names(down_table) <- paste0(rep, '.', mzgroup, '.', c('NPS_down.mean', 'NPS_down.ci_95per_low', 'NPS_down.ci_95per_high'))
		names(unaff_table) <- paste0(rep, '.', mzgroup, '.', c('NPS_unaff.mean', 'NPS_unaff.ci_95per_low', 'NPS_unaff.ci_95per_high'))

		out_summary_table <- cbind(out_summary_table, down_table, unaff_table)
	}
	out_table <- cbind(out_table, round(deltaRPKM[,3:ncol(deltaRPKM)], 2))
}
dev.off()

### Data set 3: White et al poly(A)+ timecourse

### (since poly(A)+ RNA-seq is biased by poly(A) tail length)

# 5 reps per stage, average them
stages <- gsub(".1$", "", names(white)[grep('.1$', names(white))])
for (stage in stages) {
	white[,stage] <- apply(white[,paste0(stage, '.', 1:5)], 1, mean)
}
